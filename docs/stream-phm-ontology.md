# 流式计算 + 预测性维护 — 本体定义 → AI生成

## 模式

```
本体定义 WHAT              AI 生成 HOW
─────────────             ────────────
thing_model.rules[]       gen_statem guard clauses
stream_tasks.yaml         EdgeStreamEngine Python
phm_rules.json            PHM model + SQL window

跟物模型一样的闭环:
  定义规则 → AI生成代码 → 编译部署 → 日志验证 → 修本体(不改代码)
```

## 流式计算 — 15种算法本体

```
本体定义 (stream_ontology.json):
{
  "stream_tasks": [
    {
      "id": "sudden_change_temp",
      "type": "sudden_change",
      "property": "temperature",
      "params": {"window": 10, "threshold_pct": 20, "unit": "s"},
      "trigger": "T_value_change > 20% in 10s window",
      "action": {"severity": "L1", "notify": true, "log": true}
    },
    {
      "id": "range_check_pressure",
      "type": "range_check",
      "property": "oil_pressure",
      "params": {"lo": 0.5, "hi": 3.0},
      "trigger": "V < 0.5 OR V > 3.0",
      "action": {"severity": "L2", "shutdown": true}
    },
    {
      "id": "trend_detect_vibration",
      "type": "trend_detect",
      "property": "vibration",
      "params": {"window": 60, "method": "linear_regression", "unit": "min"},
      "trigger": "slope > 0.05 AND r_squared > 0.7",
      "action": {"severity": "L1", "predictive": true}
    }
  ]
}
```

AI 自动生成:
```python
# generated by AI from stream_ontology.json

class StreamEngine:
    def sudden_change_temp(self, value, history):
        if len(history) < 10: return None
        avg = sum(history[-10:]) / 10
        change_pct = abs(value - avg) / avg * 100
        if change_pct > 20:
            return {"severity": "L1", "action": "notify", "msg": f"T change {change_pct:.1f}%"}
        return None

    def range_check_pressure(self, value):
        if value < 0.5 or value > 3.0:
            return {"severity": "L2", "action": "shutdown", "msg": f"P={value} out of [0.5,3.0]"}
        return None

    def trend_detect_vibration(self, values):
        if len(values) < 60: return None
        # linear regression on last 60 values
        n = len(values); x_mean = (n-1)/2; y_mean = sum(values)/n
        slope = sum((i-x_mean)*(v-y_mean) for i,v in enumerate(values)) / sum((i-x_mean)**2 for i in range(n))
        if slope > 0.05:
            return {"severity": "L1", "predictive": True, "msg": f"Vibration trending up: slope={slope:.3f}"}
        return None
```

## 预测性维护 — PHM 本体

```
本体定义 (phm_ontology.json):
{
  "phm_models": [
    {
      "id": "bearing_rul",
      "device_type": "pump",
      "sensors": ["vibration_x", "vibration_y", "temperature"],
      "method": "iso_10816",          <- 国际标准
      "thresholds": {
        "vibration_rms": {"good": 2.8, "alert": 4.5, "danger": 7.1},  <- ISO 10816-3
        "temperature":  {"good": 60,  "alert": 75,  "danger": 85}
      },
      "prediction": {
        "type": "remaining_useful_life",
        "method": "exponential_degradation",
        "params": {"base_rul": 2000, "degradation_rate": 0.001}
      },
      "action": {
        "alert":  {"severity": "L1", "action": "schedule_inspection"},
        "danger": {"severity": "L2", "action": "reduce_speed_50pct"},
        "critical": {"severity": "L3", "action": "emergency_stop"}
      }
    }
  ]
}
```

AI 自动生成:
```erlang
%% generated by AI from phm_ontology.json

%% gen_statem guard: pump + bearing_rul
online(cast, {data, #{vibration_rms := V, temperature := T}}, Device) ->
    case {V > 7.1, T > 85} of
        {true, _} ->  %% ISO 10816 danger zone
            action({critical, "bearing failure imminent, V=~p, T=~p", [V, T]});
        {_, true} ->  %% temp critical
            action({danger, "overtemp alarm", []});
        {false, false} when V > 4.5 ->
            action({alert, "vibration elevated, schedule inspection", []});
        _ ->
            {keep_state, Device}
    end.
```

## TDengine 预测查询

```sql
-- AI 从 phm_ontology 生成 SQL 窗口查询

-- 轴承剩余寿命预测 (exponential degradation model)
SELECT 
    device_id,
    LAST(ts) as last_ts,
    LAST(value) as current_vibration,
    LAST(value) - FIRST(value) as delta,
    COUNT(*) * 20 as runtime_hours,
    2000 - (LAST(value) - 2.8) / 0.001 as estimated_rul_hours
FROM _85ef6b7459._2de1b3e1b8
WHERE ts > NOW - 30d
  AND point_id = 'vibration_rms'
GROUP BY device_id
HAVING estimated_rul_hours < 500;
```

## 闭环保真

```
本体 (phm_ontology.json) → AI生成 → 部署 → 运行 → 日志验证

如果预测不准:
  改本体: degradation_rate: 0.001 → 0.002
  ← 不改生成的 gen_statem guard (AI重新生成)

如果ISO标准更新:
  改本体: vibration_rms.danger: 7.1 → 11.0 (ISO 10816-3新版本)
  ← AI重新生成 thresholds

如果新传感器加入:
  改本体: sensors: +["oil_quality"]
  ← AI生成新的 check_value + TDengine schema
```

## 统一到 thing_model.json

```json
{
  "properties": [
    {"identifier": "temperature", "dataType": "float", "dataForm": {...}}
  ],
  "rules": [
    {"id": "R1", "when": {"property": "temperature", "op": ">", "value": 75},
     "then": {"state": "alarm", "severity": "L1"}},
    
    {"id": "R2", "type": "stream", "algorithm": "sudden_change",
     "when": {"property": "temperature", "window": 10, "threshold": 20, "unit": "s"},
     "then": {"severity": "L1", "action": "notify"}},
    
    {"id": "R3", "type": "stream", "algorithm": "range_check",
     "when": {"property": "oil_pressure", "lo": 0.5, "hi": 3.0},
     "then": {"severity": "L2", "action": "shutdown"}},
    
    {"id": "R4", "type": "phm", "algorithm": "trend_detect",
     "when": {"property": "vibration_rms", "window": 60, "method": "linear"},
     "then": {"severity": "L1", "predictive": true}},
    
    {"id": "R5", "type": "phm", "algorithm": "iso_10816",
     "when": {"property": "vibration_rms", "standard": "ISO10816-3"},
     "then": {"good": 2.8, "alert": 4.5, "danger": 7.1}}
  ]
}
```

```
同一个 thing_model.json:
  R1: 简单阈值规则  → gen_statem guard
  R2: 流式计算规则  → EdgeStreamEngine
  R3: 流式范围检查  → EdgeStreamEngine
  R4: 趋势检测      → PHM Engine + SQL window
  R5: ISO标准       → PHM Engine + 国标阈值

AI 读取 type 字段, 分发生成不同目标代码。
本体不关心"谁来执行"——只定义"什么条件触发什么动作"。
```

## 跟物模型统一的模式

```
物模型规则:          T > 75 → alarm L1           ← 确定性
流式计算:            ΔT/10s > 20% → notify L1    ← 确定性
预测维护:            V_trend > 0.05 → RUL < 500h ← 确定性

全部是确定性规则，只是复杂度递增。
本体定义规则，AI 生成确定代码，生产环境不跑 AI 推理。
```
