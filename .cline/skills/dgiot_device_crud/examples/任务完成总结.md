## 技术实现

### 1. 治具定位地址查询关联
```erlang
create_all() ->
    %% 步骤1: 查询治具的定位地址信息
    case query_fixture_workstation_info() of
        {ok, FixtureWorkstations} ->
            %% 步骤2: 合并治具查询结果和预设工位信息
            Workstations = merge_workstation_info(FixtureWorkstations),
            %% 步骤3: 批量创建设备
            batch_create(Workstations);
        {error, Reason} ->
            %% 治具查询失败时使用预设工位信息
            batch_create(PresetWorkstations)
    end.
```
